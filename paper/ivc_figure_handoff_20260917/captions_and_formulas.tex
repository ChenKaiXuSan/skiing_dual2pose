\[
P^L,P^R \in \mathbb{R}^{T\times J\times 3},\qquad
\tilde P^{R\to L}=\mathcal A(\tilde P^R,\tilde P^L),\quad
\tilde P^{L\to R}=\mathcal A(\tilde P^L,\tilde P^R).
\]
\[
D^L=\tilde P^{R\to L}-\tilde P^L,\qquad
V^L_t=\tilde P^L_t-\tilde P^L_{t-1},\quad V^L_1=0.
\]
\[
X^L=[\tilde P^L,\tilde P^{R\to L},D^L,V^L,r^{R\to L}],\qquad
X^R=[\tilde P^R,\tilde P^{L\to R},D^R,V^R,r^{L\to R}].
\]
\[
3+3+3+3+3=15,\qquad H^L=\phi(X^L),\quad H^R=\phi(X^R).
\]
\[
Z=[\bar H^L,\bar H^R],\qquad
\alpha=\sigma(g_\theta(Z)),\quad \Delta P=h_\theta(Z),\quad
\alpha\in[0,1]^{T\times J\times1}.
\]
\[
B^L=\tfrac12(\tilde P^L+\tilde P^{R\to L}),\qquad
B^R=\tfrac12(\tilde P^R+\tilde P^{L\to R}).
\]
\[
P^{base}=\alpha B^L+(1-\alpha)B^R,\qquad
P^F=P^{base}+\Delta P.
\]

% Figure 2 caption:
% Feature construction and reliability-gated fusion. For each joint and frame,
% the own-view canonical pose, aligned opposite-view pose, cross-view difference,
% temporal first difference, and sequence-level rotation vector are concatenated
% into a 15-dimensional feature. A shared joint encoder, bidirectional cross-view
% attention, and temporal refinement produce features for joint-frame gating
% and residual correction. The final output combines the two base candidates
% using alpha and 1-alpha and adds the predicted correction Delta P. The gate
% heatmap is measured on a fixed Unity sample; attention connections are
% schematic. The right stream uses the same feature construction with the view
% roles reversed.
